51:         BigDecimal currentPrice = basePrice;
52: 
53:         // 1. Sconto promozionale
54:         if (promoDiscount > 0) {
55:             double promoFactor = 1.0 - (promoDiscount / 100.0);
56:             currentPrice = currentPrice.multiply(BigDecimal.valueOf(promoFactor));
57:         }
58: 
59:         // 2. Sconto sull'usato (20% se targa presente)
60:         if (licencePlate != null && !licencePlate.trim().isEmpty()) {

65:         int currentYear = Year.now().getValue();
66:         int yearsOld = Math.max(0, currentYear - year);
67:         if (yearsOld > 0) {
68:             double yearFactor = Math.pow(0.95, yearsOld);
69:             currentPrice = currentPrice.multiply(BigDecimal.valueOf(yearFactor));
70:         }
71: 
72:         // 4. Ammortamento chilometri (5% ogni 20.000 km)
73:         if (km > 0) {
74:             double kmExponent = (double) km / 20000.0;
75:             double kmFactor = Math.pow(0.95, kmExponent);
76:             currentPrice = currentPrice.multiply(BigDecimal.valueOf(kmFactor));
